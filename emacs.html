(defun jmn-connect-to-repositories ()
  (require 'package)
  (setq package-archives '(("melpa" . "https://melpa.org/packages/")
                           ("elpa" . "https://elpa.gnu.org/packages/")))

  (if (version< emacs-version "29")
      (progn
        (setq package-check-signature nil)  ;; bugfix: keys are not installed
        (setq gnutls-algorithm-priority "NORMAL:-VERS-TLS1.3")))  ;; to load melpa

  (package-initialize)

  (unless package-archive-contents
    (package-refresh-contents))

  (unless (package-installed-p 'use-package)
    (package-install 'use-package));; Initialize use-package on non-Linux platforms

  (require 'use-package)
  (setq use-package-always-ensure t) ; no need for :ensure t for each package.
  (setq use-package-verbose t)) ; log configure/loading messages in *Messages*

(defun jmn-use-package-if-available (orig name &rest args)
  "Around advice for `use-package' on offline systems.
Expand the block only if the library NAME is on disk (built in, or installed in
~/.emacs.d/elpa); otherwise expand to nothing so nothing is downloaded."
  (when (locate-library (symbol-name name))
    (apply orig name args)))

(if (eq jmn-system 'connected)
    (jmn-connect-to-repositories) ;; connect to repos and use use-package macros to config below
  (package-initialize)            ;; activate anything installed by hand in ~/.emacs.d/elpa
  (if (require 'use-package nil t) ;; built in on 29+, installed on pure-plus systems
      (progn
        (setq use-package-always-ensure nil)
        (advice-add 'use-package :around #'jmn-use-package-if-available))
    (defmacro use-package (&rest _)))) ;; no use-package available: do nothing

Pure-plus

On offline systems (pure and pure-plus) use-package blocks are only expanded when the library is already on disk (see jmn-use-package-if-available above). This means:

  • pure-plus packages get the same configuration as on connected systems, there is no separate enable list to keep in sync;
  • built-in libraries configured with use-package (e.g. sh-script) are configured on every system;
  • every other block expands to nothing.

To set up a pure-plus system, run jmn-install-plus-packages on a connected machine and copy ~/.emacs.d/elpa over. If a plus package fails to load, check it is in ~/.emacs.d/elpa. 

(defun jmn-install-plus-packages()
  (interactive)
  (jmn-connect-to-repositories)
  ;;install for all versions
  (dolist (pluslist jmn-plus-packages)
    (package-install pluslist 1))
  ;; install if version 27 and above
  (if (version<= "27" emacs-version)
      (progn
        (setq package-install-upgrade-built-in t)
        (dolist (pluslist jmn-plus-packages-27)
          (package-install pluslist 1))
        (unload-feature 'seq t)
        (require 'seq))))

(defun bury-compile-buffer-if-successful (buffer string)
 "Bury a compilation buffer if succeeded without warnings "
 (when (and
         (buffer-live-p buffer)
         (string-match "compilation" (buffer-name buffer))
         (string-match "finished" string)
         (not
          (with-current-buffer buffer
            (goto-char (point-min))
